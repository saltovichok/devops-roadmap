1.Git:
	                    3 директории и 2 отката   


	Рабочая директория --git add--> Индекс --git commit--> Репозиторий
        <------git restore--------  <--git restore --staged--



	Основные команды(перед ними всегда "git" держите в курсе):
	
		"init" - инициализирует репозиторий внутри каталога
		
		"status" - показывает состояние репозитория и файлы которые как либо изменялись и добавлялись в индекс

		"add" - добавляет файлы в индекс

		"commit" - делает снимок состояния репозитория

		"log" - показывает прошлые коммиты и их состояние(можно выводить подробнее с помощью --oneline --all --graph)

		"diff" - показывает различие между файлами в репозитории

		"restore" - откатывает изменения из индекса в рабочую директорию

		"revert" - откатывает изменения коммита путем создание нового коммита
		
		"branch" - инстреумент управления ветками

		"switch" - используется для перемещения(и создания с помощью "-c" веток) между ветками

		"merge" - объединяет история коммитов из разных веток путем склейки

		"tag" - создает,удаляет и просматривает теги - метки указывающие на коммиты в истории

		"clone" - клонирует репозиторий

		"fetch" - скачивает обновление репозитория из удаленного репозитория(Github) но не меняет ничего в файлах

		"pull" - так же как и fetch только он уже сразу объединяет изменения с локальной веткой
		
		"push" - отправляет коммиты в Удаленный Репозиторий



	
2.Linux basics:
	
	
	"pwd" - указывает путь в каталог в котором вы находитесь

	"ls" - показывает файлы в папке(файл запрещенные .gitignore не показываются)

	"mkdir" - создает папку/каталог/директорию(кому как привычнее)

	"cp" - копирует файлы

	"mv" - перемещает либо переименовывает файлы

	"rm" - удаляет файлы без возможности возврата(нет корзины)

	"find" - ищет файлы 

	"tail" - выводит конец (по умолчанию 10 последнийх строк но если указывать "-n <число> будет ваше число"

	"-f" - опция используемая для применения действий без каких либо подтверждений

	"which/type":
		"which" - показывает где файл
		"type" - расказывает о файле

	"apt" - команда для установки пакетов

	"ln" - команда для создания ссылок между файлами(жестких и символьных)
